class FormationAnalyzer:

    async def recognize_formation(self, player_positions: List[Tuple[float, float]],

        """Recognize team formation from player positions"""
        try:
            if len(player_positions) < 8:
                return None
            
            # Normalize positions
            positions_array = np.array(player_positions)
            normalized_positions = self.scaler.fit_transform(positions_array)
            
            # Calculate tactical shape metrics
            tactical_shape = await self._calculate_tactical_shape(normalized_positions)
            
            # Match against formation templates
            formation_match = await self._match_formation_template(normalized_positions)
            
            # Calculate formation metrics
            compactness = self._calculate_compactness(positions_array)
            width = self._calculate_width(positions_array) 
            depth = self._calculate_depth(positions_array)
            
            return Formation(
                formation_type=formation_match['type'],
                confidence=formation_match['confidence'],
                players_positions=player_positions,
                timestamp=timestamp,
                tactical_shape=tactical_shape,
                compactness=compactness,
                width=width,
                depth=depth
            )
            
        except Exception as e:
            logging.error(f"Error recognizing formation: {e}")
            return None

    async def _match_formation_template(self, positions: np.ndarray) -> Dict[str, Any]:
        """Match player positions against formation templates"""
        best_match = {'type': FormationType.CUSTOM, 'confidence': 0.0}
        
        for formation_type, template in self.formation_templates.items():
            if len(positions) == len(template):
                # Calculate distance-based similarity
                similarity = await self._calculate_formation_similarity(positions, template)
                
                if similarity > best_match['confidence']:
                    best_match = {
                        'type': formation_type,
                        'confidence': similarity
                    }
        
        return best_match
    
    async def _calculate_formation_similarity(self, positions: np.ndarray, 
                                           template: np.ndarray) -> float:
        """Calculate similarity between current positions and formation template"""
        try:
            # Use Hungarian algorithm for optimal assignment
            from scipy.optimize import linear_sum_assignment
            
            # Calculate distance matrix
            cost_matrix = np.zeros((len(positions), len(template)))
            for i, pos in enumerate(positions):
                for j, temp_pos in enumerate(template):
                    cost_matrix[i, j] = euclidean(pos, temp_pos)
            
            # Find optimal assignment
            row_indices, col_indices = linear_sum_assignment(cost_matrix)
            total_cost = cost_matrix[row_indices, col_indices].sum()
            
            # Convert cost to similarity (0-1 scale)
            max_possible_cost = len(positions) * np.sqrt(2)  # Maximum distance in normalized space
            similarity = 1.0 - (total_cost / max_possible_cost)
            
            return max(0.0, similarity)
            
        except Exception as e:
            logging.error(f"Error calculating formation similarity: {e}")
            return 0.0

class HeatMapGenerator:
    """Heat map generation and visualization"""
    
    def __init__(self):
        self.pitch_dimensions = (105, 68)  # Standard pitch dimensions in meters
        self.grid_resolution = (50, 34)   # Grid resolution for heat maps
        
    async def generate_individual_heat_map(self, player_data: List[Dict], 
                                         player_id: str) -> Dict[str, Any]:
        """Generate heat map for individual player"""
        try:
            # Extract positions
            positions = [(d['x'], d['y']) for d in player_data if 'x' in d and 'y' in d]
            
            if not positions:
                return {'error': 'No position data available'}
            
            # Create heat map grid
            heat_map_grid = self._create_heat_map_grid(positions)
            
            # Calculate zone occupancy
            zone_occupancy = self._calculate_zone_occupancy(positions)
            
            # Calculate movement patterns
            movement_patterns = await self._analyze_movement_patterns(positions)
            
            # Generate heat map visualization data
            visualization_data = self._prepare_heat_map_visualization(heat_map_grid)
            
            return {
                'player_id': player_id,
                'heat_map_grid': heat_map_grid.tolist(),
                'zone_occupancy': zone_occupancy,
                'movement_patterns': movement_patterns,
                'visualization_data': visualization_data,
                'total_positions': len(positions),
                'coverage_area': self._calculate_coverage_area(positions)
            }
            
        except Exception as e:
            logging.error(f"Error generating individual heat map: {e}")
            return {'error': str(e)}
    
    def _create_heat_map_grid(self, positions: List[Tuple[float, float]]) -> np.ndarray:
        """Create heat map grid from player positions"""
        try:
            # Create grid
            x_bins = np.linspace(0, self.pitch_dimensions[0], self.grid_resolution[0] + 1)
            y_bins = np.linspace(0, self.pitch_dimensions[1], self.grid_resolution[1] + 1)
            
            # Extract x, y coordinates
            x_coords = [pos[0] for pos in positions]
            y_coords = [pos[1] for pos in positions]
            
            # Create 2D histogram
            heat_map, _, _ = np.histogram2d(x_coords, y_coords, bins=[x_bins, y_bins])
            
            # Apply Gaussian smoothing
            from scipy.ndimage import gaussian_filter
            smoothed_heat_map = gaussian_filter(heat_map, sigma=1.0)
            
            # Normalize
            if smoothed_heat_map.max() > 0:
                smoothed_heat_map = smoothed_heat_map / smoothed_heat_map.max()
            
            return smoothed_heat_map
            
        except Exception as e:
            logging.error(f"Error creating heat map grid: {e}")
            return np.zeros(self.grid_resolution)

class PassAnalyzer:
    """Pass detection and analysis"""
    
    def __init__(self):
        self.min_pass_distance = 3.0  # Minimum distance for a pass (meters)
        self.max_pass_distance = 50.0  # Maximum distance for a pass (meters)
        self.pass_duration_threshold = 3.0  # Maximum time for a pass (seconds)
        
    async def detect_passes(self, tracking_data: List[Dict]) -> List[PassEvent]:
        """Detect pass events from tracking data"""
        try:
            pass_events = []
            
            # Sort tracking data by timestamp
            sorted_data = sorted(tracking_data, key=lambda x: x.get('timestamp', 0))
            
            # Group data by team
            team_data = {}
            for data_point in sorted_data:
                team_id = data_point.get('team_id')
                if team_id not in team_data:
                    team_data[team_id] = []
                team_data[team_id].append(data_point)
            
            # Detect passes for each team
            for team_id, team_points in team_data.items():
                team_passes = await self._detect_team_passes(team_points)
                pass_events.extend(team_passes)
            
            return pass_events
            
        except Exception as e:
            logging.error(f"Error detecting passes: {e}")
            return []
    
    async def _detect_team_passes(self, team_data: List[Dict]) -> List[PassEvent]:
        """Detect passes within a team"""
        passes = []
        
        try:
            # Simple ball possession-based pass detection
            ball_positions = [d for d in team_data if d.get('object_type') == 'ball']
            
            for i in range(len(ball_positions) - 1):
                current_ball = ball_positions[i]
                next_ball = ball_positions[i + 1]
                
                # Calculate ball movement
                distance = self._calculate_distance(
                    (current_ball['x'], current_ball['y']),
                    (next_ball['x'], next_ball['y'])
                )
                
                time_diff = next_ball['timestamp'] - current_ball['timestamp']
                
                # Check if this looks like a pass
                if (self.min_pass_distance <= distance <= self.max_pass_distance and 
                    time_diff <= self.pass_duration_threshold):
                    
                    # Find nearest players at start and end
                    passer = await self._find_nearest_player(
                        team_data, current_ball['timestamp'], 
                        (current_ball['x'], current_ball['y'])
                    )
                    
                    receiver = await self._find_nearest_player(
                        team_data, next_ball['timestamp'],
                        (next_ball['x'], next_ball['y'])
                    )
                    
                    if passer and receiver and passer['player_id'] != receiver['player_id']:
                        pass_event = PassEvent(
                            passer_id=passer['player_id'],
                            receiver_id=receiver['player_id'],
                            start_position=(current_ball['x'], current_ball['y']),
                            end_position=(next_ball['x'], next_ball['y']),
                            timestamp=datetime.fromtimestamp(current_ball['timestamp']),
                            success=True,  # Will be refined with more analysis
                            pass_type=self._classify_pass_type(distance),
                            distance=distance,
                            angle=self._calculate_pass_angle(
                                (current_ball['x'], current_ball['y']),
                                (next_ball['x'], next_ball['y'])
                            ),
                            pressure=0.0  # Will be calculated separately
                        )
                        
                        passes.append(pass_event)
            
            return passes
            
        except Exception as e:
            logging.error(f"Error detecting team passes: {e}")
            return []
    
    def _calculate_distance(self, pos1: Tuple[float, float], 
                          pos2: Tuple[float, float]) -> float:
        """Calculate Euclidean distance between two positions"""
        return np.sqrt((pos1[0] - pos2[0])**2 + (pos1[1] - pos2[1])**2)
    
    def _classify_pass_type(self, distance: float) -> str:
        """Classify pass type based on distance"""
        if distance < 10:
            return "short"
        elif distance < 25:
            return "medium"
        else:
            return "long"
    
    def _calculate_pass_angle(self, start_pos: Tuple[float, float], 
                            end_pos: Tuple[float, float]) -> float:
        """Calculate pass angle in degrees"""
        dx = end_pos[0] - start_pos[0]
        dy = end_pos[1] - start_pos[1]
        angle = np.arctan2(dy, dx) * 180 / np.pi
        return angle

class DefensiveAnalyzer:
    """Defensive pattern analysis and offside detection"""
    
    def __init__(self):
        self.offside_threshold = 1.0  # Meters threshold for offside detection
        
    async def analyze_defensive_line(self, tracking_data: List[Dict]) -> Dict[str, Any]:
        """Analyze defensive line positioning and movement"""
        try:
            defensive_lines = []
            
            # Group data by timestamp
            time_groups = {}
            for data_point in tracking_data:
                timestamp = data_point.get('timestamp')
                if timestamp not in time_groups:
                    time_groups[timestamp] = []
                time_groups[timestamp].append(data_point)
            
            # Analyze defensive line for each timestamp
            for timestamp, data_points in time_groups.items():
                # Separate by team
                teams = {}
                for dp in data_points:
                    team_id = dp.get('team_id')
                    if team_id and dp.get('object_type') == 'player':
                        if team_id not in teams:
                            teams[team_id] = []
                        teams[team_id].append(dp)
                
                # Analyze defensive line for each team
                for team_id, team_players in teams.items():
                    if len(team_players) >= 4:  # Need minimum defenders
                        defensive_line = await self._calculate_defensive_line(team_players)
                        if defensive_line:
                            defensive_line['timestamp'] = timestamp
                            defensive_line['team_id'] = team_id
                            defensive_lines.append(defensive_line)
            
            # Calculate defensive line statistics
            statistics = await self._calculate_defensive_line_statistics(defensive_lines)
            
            return {
                'defensive_lines': defensive_lines,
                'statistics': statistics
            }
            
        except Exception as e:
            logging.error(f"Error analyzing defensive line: {e}")
            return {'error': str(e)}
    
    async def _calculate_defensive_line(self, players: List[Dict]) -> Optional[Dict]:
        """Calculate defensive line position and metrics"""
        try:
            # Extract defender positions (assume players closest to own goal)
            positions = [(p['x'], p['y']) for p in players if 'x' in p and 'y' in p]
            
            if len(positions) < 4:
                return None
            
            # Sort by x-coordinate (defensive position)
            positions.sort(key=lambda p: p[0])
            
            # Take the 4 most defensive players
            defenders = positions[:4]
            
            # Calculate defensive line metrics
            x_positions = [d[0] for d in defenders]
            y_positions = [d[1] for d in defenders]
            
            defensive_line_x = np.mean(x_positions)
            defensive_line_y = np.mean(y_positions)
            
            # Calculate compactness (standard deviation of y positions)
            compactness = np.std(y_positions)
            
            # Calculate line straightness (variance from average x position)
            straightness = 1.0 / (1.0 + np.var(x_positions))
            
            return {
                'position': (defensive_line_x, defensive_line_y),
                'compactness': compactness,
                'straightness': straightness,
                'width': max(y_positions) - min(y_positions),
                'depth': max(x_positions) - min(x_positions),
                'defender_positions': defenders
            }
            
        except Exception as e:
            logging.error(f"Error calculating defensive line: {e}")
            return None
    
    async def detect_offside(self, tracking_data: List[Dict]) -> List[Dict]:
        """Detect potential offside situations"""
        try:
            offside_events = []
            
            # Group data by timestamp
            time_groups = {}
            for data_point in tracking_data:
                timestamp = data_point.get('timestamp')
                if timestamp not in time_groups:
                    time_groups[timestamp] = []
                time_groups[timestamp].append(data_point)
            
            # Check for offside at each timestamp
            for timestamp, data_points in time_groups.items():
                # Separate teams and get player positions
                team_positions = {}
                ball_position = None
                
                for dp in data_points:
                    if dp.get('object_type') == 'ball':
                        ball_position = (dp['x'], dp['y'])
                    elif dp.get('object_type') == 'player':
                        team_id = dp.get('team_id')
                        if team_id:
                            if team_id not in team_positions:
                                team_positions[team_id] = []
                            team_positions[team_id].append(dp)
                
                # Check offside for each team
                if ball_position and len(team_positions) == 2:
                    teams = list(team_positions.keys())
                    
                    # Check team 1 attacking against team 2
                    offside_check_1 = await self._check_team_offside(
                        team_positions[teams[0]], team_positions[teams[1]], 
                        ball_position, timestamp
                    )
                    
                    # Check team 2 attacking against team 1
                    offside_check_2 = await self._check_team_offside(
                        team_positions[teams[1]], team_positions[teams[0]], 
                        ball_position, timestamp
                    )
                    
                    offside_events.extend(offside_check_1)
                    offside_events.extend(offside_check_2)
            
            return offside_events
            
        except Exception as e:
            logging.error(f"Error detecting offside: {e}")
            return []
    
    async def _check_team_offside(self, attacking_team: List[Dict], 
                                defending_team: List[Dict], 
                                ball_position: Tuple[float, float], 
                                timestamp: float) -> List[Dict]:
        """Check offside for attacking team against defending team"""
        offside_events = []
        
        try:
            # Find second-to-last defender position
            defender_x_positions = [p['x'] for p in defending_team if 'x' in p]
            if len(defender_x_positions) < 2:
                return offside_events
            
            defender_x_positions.sort(reverse=True)  # Sort descending (towards goal)
            second_last_defender_x = defender_x_positions[1]
            
            # Check each attacking player
            for attacker in attacking_team:
                if 'x' in attacker and 'y' in attacker:
                    # Player is in offside position if they are ahead of second-to-last defender
                    # and ball, and in opponent's half
                    if (attacker['x'] > second_last_defender_x + self.offside_threshold and
                        attacker['x'] > ball_position[0] and
                        attacker['x'] > 52.5):  # Opponent's half (assuming 105m pitch)
                        
                        offside_events.append({
                            'timestamp': timestamp,
                            'player_id': attacker.get('player_id'),
                            'player_position': (attacker['x'], attacker['y']),
                            'offside_line': second_last_defender_x,
                            'ball_position': ball_position,
                            'severity': attacker['x'] - second_last_defender_x
                        })
            
            return offside_events
            
        except Exception as e:
            logging.error(f"Error checking team offside: {e}")
            return []
