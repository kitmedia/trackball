## Team Settings Implementation ⏳

### Comprehensive Settings Management System
```typescript
// Team Settings Manager Component
import React, { useState, useCallback, useEffect } from 'react';
import { TeamSettingsState, SettingsUpdate, ValidationResult } from './types';

export const TeamSettingsManager: React.FC<TeamSettingsProps> = ({
  teamId,
  currentUser,
  onSettingsUpdate,
  onValidationError
}) => {
  const [settingsState, setSettingsState] = useState<TeamSettingsState>({
    activeSettingsTab: 'profile',
    teamProfile: null,
    analysisDefaults: null,
    notificationPreferences: null,
    storageConfiguration: null,
    pendingChanges: [],
    settingsValidation: { isValid: true, errors: [] }
  });
  
  const [isLoading, setIsLoading] = useState(false);
  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);

  // Load team settings on component mount
  useEffect(() => {
    const loadTeamSettings = async () => {
      setIsLoading(true);
      try {
        const settings = await fetchTeamSettings(teamId);
        setSettingsState(prev => ({
          ...prev,
          teamProfile: settings.profile,
          analysisDefaults: settings.analysisDefaults,
          notificationPreferences: settings.notifications,
          storageConfiguration: settings.storage,
          customFields: settings.customFields,
          usageAnalytics: settings.analytics
        }));
      } catch (error) {
        console.error('Failed to load team settings:', error);
        showErrorMessage('Failed to load team settings');
      } finally {
        setIsLoading(false);
      }
    };

    loadTeamSettings();
  }, [teamId]);

  // Handle settings updates with validation
  const handleSettingsUpdate = useCallback(async (
    section: string,
    updates: any
  ): Promise<void> => {
    try {
      // Validate changes
      const validation = await validateSettingsUpdate(section, updates);
      
      if (!validation.isValid) {
        setSettingsState(prev => ({
          ...prev,
          settingsValidation: validation
        }));
        onValidationError?.(validation.errors);
        return;
      }

      // Apply updates to local state
      setSettingsState(prev => ({
        ...prev,
        [section]: { ...prev[section], ...updates },
        pendingChanges: [
          ...prev.pendingChanges,
          { section, updates, timestamp: Date.now() }
        ]
      }));
      
      setHasUnsavedChanges(true);
      
    } catch (error) {
      console.error('Settings update failed:', error);
      showErrorMessage('Failed to update settings');
    }
  }, [onValidationError]);

  // Save all pending changes
  const saveAllChanges = useCallback(async (): Promise<void> => {
    if (settingsState.pendingChanges.length === 0) return;
    
    setIsLoading(true);
    try {
      // Batch update all changes
      const updateResult = await updateTeamSettings(teamId, settingsState.pendingChanges);
      
      if (updateResult.success) {
        setSettingsState(prev => ({
          ...prev,
          pendingChanges: [],
          settingsValidation: { isValid: true, errors: [] }
        }));
        
        setHasUnsavedChanges(false);
        showSuccessMessage('Settings saved successfully');
        onSettingsUpdate?.(updateResult.appliedChanges);
      } else {
        setSettingsState(prev => ({
          ...prev,
          settingsValidation: {
            isValid: false,
            errors: updateResult.errors
          }
        }));
        onValidationError?.(updateResult.errors);
      }
      
    } catch (error) {
      console.error('Failed to save settings:', error);
      showErrorMessage('Failed to save settings');
    } finally {
      setIsLoading(false);
    }
  }, [teamId, settingsState.pendingChanges, onSettingsUpdate, onValidationError]);

  // Discard pending changes
  const discardChanges = useCallback((): void => {
    setSettingsState(prev => ({
      ...prev,
      pendingChanges: [],
      settingsValidation: { isValid: true, errors: [] }
    }));
    setHasUnsavedChanges(false);
  }, []);

  // Settings tabs configuration
  const settingsTabs = [
    { id: 'profile', label: 'Team Profile', icon: '👥' },
    { id: 'analysis', label: 'Analysis Defaults', icon: '⚙️' },
    { id: 'notifications', label: 'Notifications', icon: '🔔' },
    { id: 'storage', label: 'Storage & Quota', icon: '💾' },
    { id: 'fields', label: 'Custom Fields', icon: '📝' },
    { id: 'backup', label: 'Backup & Export', icon: '📦' },
    { id: 'integrations', label: 'Integrations', icon: '🔗' }
  ];

  // Render active settings tab content
  const renderTabContent = () => {
    switch (settingsState.activeSettingsTab) {
      case 'profile':
        return (
          <TeamProfileSettings
            profile={settingsState.teamProfile}
            validation={settingsState.settingsValidation}
            onUpdate={(updates) => handleSettingsUpdate('teamProfile', updates)}
          />
        );
        
      case 'analysis':
        return (
          <AnalysisDefaultsSettings
            defaults={settingsState.analysisDefaults}
            validation={settingsState.settingsValidation}
            onUpdate={(updates) => handleSettingsUpdate('analysisDefaults', updates)}
          />
        );
        
      case 'notifications':
        return (
          <NotificationSettings
            preferences={settingsState.notificationPreferences}
            validation={settingsState.settingsValidation}
            onUpdate={(updates) => handleSettingsUpdate('notificationPreferences', updates)}
          />
        );
        
      case 'storage':
        return (
          <StorageManagement
            configuration={settingsState.storageConfiguration}
            usage={settingsState.storageUsage}
            analytics={settingsState.usageAnalytics}
            onUpdate={(updates) => handleSettingsUpdate('storageConfiguration', updates)}
          />
        );
        
      case 'fields':
        return (
          <CustomFieldBuilder
            fields={settingsState.customFields}
            onCreateField={(field) => createCustomField(field)}
            onUpdateField={(fieldId, updates) => updateCustomField(fieldId, updates)}
            onDeleteField={(fieldId) => deleteCustomField(fieldId)}
          />
        );
        
      case 'backup':
        return (
          <BackupConfiguration
            configuration={settingsState.backupConfiguration}
            onUpdate={(updates) => handleSettingsUpdate('backupConfiguration', updates)}
            onExport={(options) => exportTeamData(options)}
          />
        );
        
      case 'integrations':
        return (
          <IntegrationManager
            integrations={settingsState.integrationSettings}
            onConfigure={(integration) => configureIntegration(integration)}
            onTest={(integrationId) => testIntegration(integrationId)}
          />
        );
        
      default:
        return <div>Settings section not found</div>;
    }
  };

  return (
    <div className="team-settings-manager">
      {/* Settings header with save/discard controls */}
      <div className="settings-header">
        <h2>Team Settings</h2>
        
        {hasUnsavedChanges && (
          <div className="unsaved-changes-indicator">
            <span className="changes-count">
              {settingsState.pendingChanges.length} unsaved changes
            </span>
            <div className="change-actions">
              <button
                onClick={discardChanges}
                className="btn btn-secondary"
                disabled={isLoading}
              >
                Discard
              </button>
              <button
                onClick={saveAllChanges}
                className="btn btn-primary"
                disabled={isLoading}
              >
                {isLoading ? 'Saving...' : 'Save All Changes'}
              </button>
            </div>
          </div>
        )}
      </div>

      {/* Validation errors display */}
      {!settingsState.settingsValidation.isValid && (
        <div className="validation-errors">
          <h4>Please fix the following errors:</h4>
          <ul>
            {settingsState.settingsValidation.errors.map((error, index) => (
              <li key={index} className="error-item">
                {error.field}: {error.message}
              </li>
            ))}
          </ul>
        </div>
      )}

      <div className="settings-content">
        {/* Settings navigation tabs */}
        <div className="settings-navigation">
          {settingsTabs.map(tab => (
            <button
              key={tab.id}
              onClick={() => setSettingsState(prev => ({
                ...prev,
                activeSettingsTab: tab.id
              }))}
              className={`nav-tab ${
                settingsState.activeSettingsTab === tab.id ? 'active' : ''
              }`}
            >
              <span className="tab-icon">{tab.icon}</span>
              <span className="tab-label">{tab.label}</span>
              {settingsState.pendingChanges.some(c => c.section.includes(tab.id)) && (
                <span className="pending-indicator">•</span>
              )}
            </button>
          ))}
        </div>

        {/* Active tab content */}
        <div className="settings-panel">
          {isLoading ? (
            <div className="loading-state">

// API Functions
async function fetchTeamSettings(teamId: string): Promise<any> {
  const response = await fetch(`/api/teams/${teamId}/settings`);
  if (!response.ok) {
    throw new Error('Failed to fetch team settings');
  }
  return response.json();
}

async function updateTeamSettings(
  teamId: string,
  changes: SettingsChange[]
): Promise<any> {
  const response = await fetch(`/api/teams/${teamId}/settings`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ changes })
  });
  
  if (!response.ok) {
    throw new Error('Failed to update team settings');
  }
  
  return response.json();
}

async function validateSettingsUpdate(
  section: string,
  updates: any
): Promise<ValidationResult> {
  const response = await fetch('/api/settings/validate', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ section, updates })
  });
  
  return response.json();
}

async function uploadTeamLogo(file: File): Promise<{ logoUrl: string }> {
  const formData = new FormData();
  formData.append('logo', file);
  
  const response = await fetch('/api/teams/logo', {
    method: 'POST',
    body: formData
  });
  
  if (!response.ok) {
    throw new Error('Failed to upload team logo');
  }
  
  return response.json();
}
```
