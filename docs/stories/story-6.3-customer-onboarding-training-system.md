# Backend: Customer Onboarding System
from typing import Dict, List, Optional, Any, Union
import asyncio
import json
import logging
from datetime import datetime, timedelta
from dataclasses import dataclass, asdict
from enum import Enum
import uuid
from pathlib import Path
import aioredis
from sqlalchemy import Column, String, Integer, DateTime, Boolean, Text, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import relationship

Base = declarative_base()

class OnboardingStepStatus(Enum):
    NOT_STARTED = "not_started"
    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"
    SKIPPED = "skipped"

class UserRole(Enum):
    COACH = "coach"
    ANALYST = "analyst"
    PLAYER = "player"
    ADMIN = "admin"

@dataclass
class OnboardingStep:
    id: str
    name: str
    description: str
    order: int
    required: bool
    role_specific: List[UserRole]
    estimated_time: int  # minutes
    dependencies: List[str]
    resources: List[str]
    completion_criteria: Dict[str, Any]

@dataclass
class UserProgress:
    user_id: str
    step_id: str
    status: OnboardingStepStatus
    started_at: Optional[datetime]
    completed_at: Optional[datetime]
    attempts: int
    time_spent: int  # seconds
    feedback: Optional[str]

class OnboardingStep(Base):
    __tablename__ = 'onboarding_steps'
    
    id = Column(String, primary_key=True)
    name = Column(String, nullable=False)
    description = Column(Text)
    order = Column(Integer, nullable=False)
    required = Column(Boolean, default=True)
    role_specific = Column(Text)  # JSON array of roles
    estimated_time = Column(Integer)
    dependencies = Column(Text)  # JSON array of step IDs
    resources = Column(Text)  # JSON array of resource URLs
    completion_criteria = Column(Text)  # JSON object
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class UserOnboardingProgress(Base):
    __tablename__ = 'user_onboarding_progress'
    
    id = Column(String, primary_key=True, default=lambda: str(uuid.uuid4()))
    user_id = Column(String, nullable=False)
    step_id = Column(String, ForeignKey('onboarding_steps.id'), nullable=False)
    status = Column(String, nullable=False)
    started_at = Column(DateTime)
    completed_at = Column(DateTime)
    attempts = Column(Integer, default=0)
    time_spent = Column(Integer, default=0)
    feedback = Column(Text)
    
    step = relationship("OnboardingStep")

class CustomerOnboardingManager:
    """Comprehensive customer onboarding and training system"""
    
    def __init__(self, db_session, redis_client: aioredis.Redis):
        self.db = db_session
        self.redis = redis_client
        self.logger = logging.getLogger(__name__)
        
        # Onboarding components
        self.tutorial_manager = TutorialManager()
        self.help_system = ContextualHelpSystem()
        self.knowledge_base = KnowledgeBaseManager()
        self.webinar_system = WebinarManager()
        self.success_manager = CustomerSuccessManager()
        self.forum_manager = CommunityForumManager()
        
        # Onboarding configuration
        self.onboarding_steps = self._load_onboarding_steps()
        self.role_specific_paths = self._load_role_specific_paths()
        
    async def get_user_onboarding_progress(self, user_id: str) -> Dict[str, Any]:
        """Get comprehensive onboarding progress for user"""
        try:
            # Get user role and team info
            user_info = await self._get_user_info(user_id)
            user_role = UserRole(user_info.get('role', 'coach'))
            
            # Get applicable onboarding steps
            applicable_steps = await self._get_applicable_steps(user_role)
            
            # Get user progress
            progress_records = self.db.query(UserOnboardingProgress).filter(
                UserOnboardingProgress.user_id == user_id
            ).all()
            
            progress_map = {p.step_id: p for p in progress_records}
            
            # Build progress summary
            steps_progress = []
            total_steps = len(applicable_steps)
            completed_steps = 0
            total_time_spent = 0
            
            for step in applicable_steps:
                progress = progress_map.get(step.id)
                
                step_info = {
                    'step_id': step.id,
                    'name': step.name,
                    'description': step.description,
                    'order': step.order,
                    'required': step.required,
                    'estimated_time': step.estimated_time,
                    'status': progress.status if progress else OnboardingStepStatus.NOT_STARTED.value,
                    'started_at': progress.started_at.isoformat() if progress and progress.started_at else None,
                    'completed_at': progress.completed_at.isoformat() if progress and progress.completed_at else None,
                    'time_spent': progress.time_spent if progress else 0,
                    'attempts': progress.attempts if progress else 0,
                    'resources': json.loads(step.resources) if step.resources else []
                }
                
                steps_progress.append(step_info)
                
                if progress and progress.status == OnboardingStepStatus.COMPLETED.value:
                    completed_steps += 1
                
                if progress:
                    total_time_spent += progress.time_spent
            
            # Calculate overall progress
            completion_percentage = (completed_steps / total_steps * 100) if total_steps > 0 else 0
            
            # Get next recommended step
            next_step = await self._get_next_step(user_id, applicable_steps, progress_map)
            
            # Get personalized recommendations
            recommendations = await self._get_personalized_recommendations(user_id, user_role)
            
            return {
                'user_id': user_id,
                'user_role': user_role.value,
                'overall_progress': {
                    'completion_percentage': completion_percentage,
                    'completed_steps': completed_steps,
                    'total_steps': total_steps,
                    'total_time_spent': total_time_spent,
                    'estimated_remaining_time': sum(
                        step.estimated_time * 60 for step in applicable_steps 
                        if step.id not in progress_map or 
                        progress_map[step.id].status != OnboardingStepStatus.COMPLETED.value
                    )
                },
                'steps': steps_progress,
                'next_step': next_step,
                'recommendations': recommendations
            }
            
        except Exception as e:
            self.logger.error(f"Error getting onboarding progress: {e}")
            return {'error': str(e)}
    
    async def complete_onboarding_step(self, user_id: str, step_id: str, 
                                     completion_data: Optional[Dict] = None) -> Dict[str, Any]:
        """Mark onboarding step as completed"""
        try:
            # Get or create progress record
            progress = self.db.query(UserOnboardingProgress).filter(
                UserOnboardingProgress.user_id == user_id,
                UserOnboardingProgress.step_id == step_id
            ).first()
            
            if not progress:
                progress = UserOnboardingProgress(
                    user_id=user_id,
                    step_id=step_id,
                    status=OnboardingStepStatus.IN_PROGRESS.value,
                    started_at=datetime.utcnow(),
                    attempts=1
                )
                self.db.add(progress)
            
            # Update progress
            progress.status = OnboardingStepStatus.COMPLETED.value
            progress.completed_at = datetime.utcnow()
            progress.attempts += 1
            
            if completion_data:
                progress.feedback = json.dumps(completion_data)
                progress.time_spent += completion_data.get('time_spent', 0)
            
            self.db.commit()
            
            # Check for achievements/milestones
            achievements = await self._check_achievements(user_id)
            
            # Send completion notifications
            await self._send_completion_notifications(user_id, step_id)
            
            # Update customer success metrics
            await self.success_manager.update_onboarding_metrics(user_id, step_id)
            
            return {
                'success': True,
                'step_id': step_id,
                'status': progress.status,
                'achievements': achievements
            }
            
        except Exception as e:
            self.logger.error(f"Error completing onboarding step: {e}")
            self.db.rollback()
            return {'success': False, 'error': str(e)}
    
    async def get_next_onboarding_step(self, user_id: str) -> Optional[Dict[str, Any]]:
        """Get next recommended onboarding step"""
        try:
            # Get user info and progress
            user_info = await self._get_user_info(user_id)
            user_role = UserRole(user_info.get('role', 'coach'))
            
            applicable_steps = await self._get_applicable_steps(user_role)
            
            progress_records = self.db.query(UserOnboardingProgress).filter(
                UserOnboardingProgress.user_id == user_id
            ).all()
            
            progress_map = {p.step_id: p for p in progress_records}
            
            # Find next step
            next_step = await self._get_next_step(user_id, applicable_steps, progress_map)
            
            if next_step:
                # Get step resources and preparation info
                step_details = await self._enrich_step_details(next_step, user_id)
                return step_details
            
            return None
            
        except Exception as e:
            self.logger.error(f"Error getting next onboarding step: {e}")
            return None
    
    async def _get_applicable_steps(self, user_role: UserRole) -> List[OnboardingStep]:
        """Get onboarding steps applicable to user role"""
        try:
            all_steps = self.db.query(OnboardingStep).order_by(OnboardingStep.order).all()
            
            applicable_steps = []
            for step in all_steps:
                # Check if step is applicable to user role
                if step.role_specific:
                    role_list = json.loads(step.role_specific)
                    if user_role.value not in role_list:
                        continue
                
                applicable_steps.append(step)
            
            return applicable_steps
            
        except Exception as e:
            self.logger.error(f"Error getting applicable steps: {e}")
            return []
    
    async def _get_next_step(self, user_id: str, applicable_steps: List[OnboardingStep], 
                           progress_map: Dict[str, UserOnboardingProgress]) -> Optional[Dict]:
        """Determine next onboarding step for user"""
        try:
            for step in applicable_steps:
                progress = progress_map.get(step.id)
                
                # Skip completed steps
                if progress and progress.status == OnboardingStepStatus.COMPLETED.value:
                    continue
                
                # Check dependencies
                dependencies_met = await self._check_dependencies(step, progress_map)
                
                if dependencies_met:
                    return {
                        'step_id': step.id,
                        'name': step.name,
                        'description': step.description,
                        'estimated_time': step.estimated_time,
                        'required': step.required,
                        'resources': json.loads(step.resources) if step.resources else []
                    }
            
            return None
            
        except Exception as e:
            self.logger.error(f"Error getting next step: {e}")
            return None
    
    async def _check_dependencies(self, step: OnboardingStep, 
                                progress_map: Dict[str, UserOnboardingProgress]) -> bool:
        """Check if step dependencies are met"""
        try:
            if not step.dependencies:
                return True
            
            dependencies = json.loads(step.dependencies)
            
            for dep_step_id in dependencies:
                dep_progress = progress_map.get(dep_step_id)
                if not dep_progress or dep_progress.status != OnboardingStepStatus.COMPLETED.value:
                    return False
            
            return True
            
        except Exception as e:
            self.logger.error(f"Error checking dependencies: {e}")
            return False

class TutorialManager:
    """Video tutorial library and learning management"""
    
    def __init__(self):
        self.tutorials = {}
        self.learning_paths = {}
        self.user_progress = {}
        
    async def get_tutorial_recommendations(self, user_id: str, 
                                         context: Optional[str] = None) -> List[Dict[str, Any]]:
        """Get personalized tutorial recommendations"""
        try:
            # Get user info and progress
            user_info = await self._get_user_info(user_id)
            user_role = user_info.get('role', 'coach')
            
            # Get user's tutorial history
            user_tutorials = self.user_progress.get(user_id, {})
            
            # Get contextual tutorials if context provided
            contextual_tutorials = []
            if context:
                contextual_tutorials = await self._get_contextual_tutorials(context)
            
            # Get role-based recommendations
            role_tutorials = await self._get_role_based_tutorials(user_role)
            
            # Get next-in-path tutorials
            path_tutorials = await self._get_learning_path_tutorials(user_id)
            
            # Combine and rank recommendations
            all_recommendations = contextual_tutorials + role_tutorials + path_tutorials
            
            # Remove already completed tutorials
            filtered_recommendations = [
                tutorial for tutorial in all_recommendations
                if tutorial['id'] not in user_tutorials or 
                user_tutorials[tutorial['id']].get('completed', False) is False
            ]
            
            # Rank by relevance and user preferences
            ranked_recommendations = await self._rank_tutorials(
                filtered_recommendations, user_info, context
            )
            
            return ranked_recommendations[:10]  # Return top 10
            
        except Exception as e:
            logging.error(f"Error getting tutorial recommendations: {e}")
            return []
    
    async def track_tutorial_progress(self, user_id: str, tutorial_id: str, 
                                    progress_data: Dict[str, Any]) -> Dict[str, Any]:
        """Track user progress through tutorial"""
        try:
            if user_id not in self.user_progress:
                self.user_progress[user_id] = {}
            
            if tutorial_id not in self.user_progress[user_id]:
                self.user_progress[user_id][tutorial_id] = {
                    'started_at': datetime.now().isoformat(),
                    'progress_percentage': 0,
                    'time_spent': 0,
                    'completed': False,
                    'quiz_scores': [],
                    'bookmarks': []
                }
            
            tutorial_progress = self.user_progress[user_id][tutorial_id]
            
            # Update progress
            tutorial_progress['progress_percentage'] = progress_data.get('progress_percentage', 0)
            tutorial_progress['time_spent'] += progress_data.get('time_spent', 0)
            
            # Handle completion

                        <Box display="flex" alignItems="center" gap={1}>
                          <ScheduleIcon fontSize="small" color="action" />
                          <Typography variant="caption">
                            ~{step.estimatedTime} min
                          </Typography>
                        </Box>
                        {step.timeSpent > 0 && (
                          <Box display="flex" alignItems="center" gap={1}>
                            <Typography variant="caption" color="text.secondary">
                              Spent: {formatTime(step.timeSpent)}
                            </Typography>
                          </Box>
                        )}
                      </Box>

                      {step.resources.length > 0 && (
                        <Box mb={2}>
                          <Typography variant="caption" color="text.secondary" display="block" mb={1}>
                            Resources:
                          </Typography>
                          {step.resources.map((resource, idx) => (
                            <Chip
                              key={idx}
                              size="small"
                              label={resource.title}
                              variant="outlined"
                              sx={{ mr: 1, mb: 1 }}
                              onClick={() => window.open(resource.url, '_blank')}
                            />
                          ))}
                        </Box>
                      )}

                      <Box display="flex" gap={1}>
                        {step.status !== 'completed' && (
                          <Button
                            variant="contained"
                            size="small"
                            startIcon={<PlayIcon />}
                            onClick={() => {
                              setSelectedStep(step);
                              setStepDialogOpen(true);
                            }}
                          >
                            {step.status === 'in_progress' ? 'Continue' : 'Start'}
                          </Button>
                        )}
                        {!step.required && step.status === 'not_started' && (
                          <Button
                            variant="outlined"
                            size="small"
                            onClick={() => completeStep(step.stepId, { skipped: true })}
                          >
                            Skip
                          </Button>
                        )}
                      </Box>
                    </StepContent>
                  </Step>
                ))}
              </Stepper>
            </CardContent>
          </Card>
        </Grid>

        <Grid item xs={12} md={4}>
          {/* Recommendations */}
          <Card sx={{ mb: 3 }}>
            <CardContent>
              <Typography variant="h6" gutterBottom>
                <TipIcon sx={{ mr: 1, verticalAlign: 'middle' }} />
                Recommendations
              </Typography>
              
              <List dense>
                {onboardingProgress?.recommendations.slice(0, 5).map((rec, index) => (
                  <ListItem key={index} sx={{ px: 0 }}>
                    <ListItemIcon>
                      {rec.type === 'tutorial' && <TutorialIcon color="primary" />}
                      {rec.type === 'webinar' && <WebinarIcon color="secondary" />}
                      {rec.type === 'help_article' && <HelpIcon color="action" />}
                    </ListItemIcon>
                    <ListItemText
                      primary={rec.title}
                      secondary={rec.description}
                      primaryTypographyProps={{ variant: 'body2' }}
                      secondaryTypographyProps={{ variant: 'caption' }}
                    />
                    <Chip
                      size="small"
                      label={rec.priority}
                      color={rec.priority === 'high' ? 'error' : rec.priority === 'medium' ? 'warning' : 'default'}
                    />
                  </ListItem>
                ))}
              </List>
            </CardContent>
          </Card>

          {/* Featured Tutorials */}
          <Card>
            <CardContent>
              <Typography variant="h6" gutterBottom>
                <TutorialIcon sx={{ mr: 1, verticalAlign: 'middle' }} />
                Featured Tutorials
              </Typography>
              
              <List dense>
                {tutorials.slice(0, 3).map((tutorial) => (
                  <ListItem key={tutorial.id} sx={{ px: 0 }}>
                    <Avatar
                      src={tutorial.thumbnailUrl}
                      sx={{ width: 40, height: 40, mr: 2 }}
                    >
                      <PlayIcon />
                    </Avatar>
                    <ListItemText
                      primary={tutorial.title}
                      secondary={
                        <Box display="flex" alignItems="center" gap={1}>
                          <Typography variant="caption">
                            {tutorial.duration} min
                          </Typography>
                          <Chip
                            size="small"
                            label={tutorial.difficulty}
                            variant="outlined"
                          />
                        </Box>
                      }
                      primaryTypographyProps={{ variant: 'body2' }}
                    />
                    {tutorial.progress !== undefined && (
                      <Box sx={{ minWidth: 35 }}>
                        <Typography variant="caption" color="text.secondary">
                          {tutorial.progress}%
                        </Typography>
                      </Box>
                    )}
                  </ListItem>
                ))}
              </List>
              
              <Button
                variant="outlined"
                fullWidth
                sx={{ mt: 2 }}
                href="/tutorials"
              >
                View All Tutorials
              </Button>
            </CardContent>
          </Card>
        </Grid>
      </Grid>

      {/* Step Detail Dialog */}
      <Dialog 
        open={stepDialogOpen} 
        onClose={() => setStepDialogOpen(false)}
        maxWidth="md"
        fullWidth
      >
        <DialogTitle>
          {selectedStep?.name}
        </DialogTitle>
        <DialogContent>
          <Typography variant="body1" paragraph>
            {selectedStep?.description}
          </Typography>
          
          <Box display="flex" alignItems="center" gap={2} mb={3}>
            <Chip
              icon={<ScheduleIcon />}
              label={`~${selectedStep?.estimatedTime} minutes`}
              variant="outlined"
            />
            {selectedStep?.required && (
              <Chip label="Required" color="error" size="small" />
            )}
          </Box>

          {selectedStep?.resources && selectedStep.resources.length > 0 && (
            <Box>
              <Typography variant="subtitle2" gutterBottom>
                Resources & Materials:
              </Typography>
              <List dense>
                {selectedStep.resources.map((resource, index) => (
                  <ListItem key={index} sx={{ px: 0 }}>
                    <ListItemText
                      primary={resource.title}
                      secondary={resource.description}
                    />
                    <Button
                      size="small"
                      variant="outlined"
                      onClick={() => window.open(resource.url, '_blank')}
                    >
                      Open
                    </Button>
                  </ListItem>
                ))}
              </List>
            </Box>
          )}
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setStepDialogOpen(false)}>
            Close
          </Button>
          {selectedStep && selectedStep.status !== 'completed' && (
            <Button
              variant="contained"
              startIcon={<CheckIcon />}
              onClick={() => {
                completeStep(selectedStep.stepId);
                setStepDialogOpen(false);
              }}
            >
              Mark Complete
            </Button>
          )}
        </DialogActions>
      </Dialog>
    </Box>
  );
};
