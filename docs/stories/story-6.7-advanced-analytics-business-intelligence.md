    def _calculate_usage_score(self, usage_data: Dict[str, Any]) -> float:
        """Calculate customer usage score (0-100)"""
        # Simplified scoring algorithm
        login_frequency = usage_data.get("login_frequency", 0)
        session_duration = usage_data.get("avg_session_duration", 0)
        feature_usage = usage_data.get("features_used", 0)
        
        score = (
            min(login_frequency * 10, 40) +    # Max 40 points for login frequency
            min(session_duration / 30 * 30, 30) +  # Max 30 points for session duration
            min(feature_usage * 5, 30)         # Max 30 points for feature usage
        )
        
        return min(score, 100)
    
    def _calculate_engagement_score(self, usage_data: Dict[str, Any]) -> float:
        """Calculate customer engagement score (0-100)"""
        # Simplified engagement scoring
        video_uploads = usage_data.get("video_uploads", 0)
        analysis_sessions = usage_data.get("analysis_sessions", 0)
        team_collaboration = usage_data.get("team_collaboration", 0)
        
        score = (
            min(video_uploads * 15, 50) +      # Max 50 points for video uploads
            min(analysis_sessions * 10, 30) +  # Max 30 points for analysis
            min(team_collaboration * 20, 20)   # Max 20 points for collaboration
        )
        
        return min(score, 100)
    
    def _calculate_support_score(self, support_data: Dict[str, Any]) -> float:
        """Calculate customer support score (0-100)"""
        ticket_count = support_data.get("ticket_count", 0)
        avg_resolution_time = support_data.get("avg_resolution_time", 0)
        satisfaction_rating = support_data.get("satisfaction_rating", 5)
        
        # Lower ticket count and higher satisfaction = higher score
        score = (
            max(100 - ticket_count * 10, 0) * 0.4 +      # Fewer tickets = better
            satisfaction_rating * 20 * 0.6                # Higher satisfaction = better
        )
        
        return min(score, 100)
    
    def _calculate_payment_score(self, payment_data: Dict[str, Any]) -> float:
        """Calculate customer payment score (0-100)"""
        payment_failures = payment_data.get("payment_failures", 0)
        days_overdue = payment_data.get("days_overdue", 0)
        subscription_tier = payment_data.get("subscription_tier", "basic")
        
        # Lower failures and no overdue = higher score
        score = (
            max(100 - payment_failures * 20, 0) * 0.6 +   # Fewer failures = better
            max(100 - days_overdue * 5, 0) * 0.4          # No overdue = better
        )
        
        # Bonus for higher tiers
        tier_bonuses = {"basic": 0, "professional": 5, "enterprise": 10}
        score += tier_bonuses.get(subscription_tier, 0)
        
        return min(score, 100)
    
    async def _predict_churn_probability(self, customer_id: str, features: Dict[str, float]) -> float:
        """Predict customer churn probability using ML model"""
        try:
            # Prepare features for model
            feature_array = np.array([[
                features['usage_score'],
                features['engagement_score'],
                features['support_score'],
                features['payment_score']
            ]])
            
            # Use trained model to predict churn probability
            churn_prob = self.ml_models['churn_prediction'].predict(feature_array)[0]
            
            # Ensure probability is between 0 and 1
            return max(0, min(churn_prob, 1))
            
        except Exception as e:
            # Fallback to simple heuristic if ML model fails
            health_score = (
                features['usage_score'] + features['engagement_score'] + 
                features['support_score'] + features['payment_score']

            churn_rate = 0.052  # 5.2% monthly

            gross_margin = 0.85  # 85% margin
            clv = (avg_monthly_revenue * gross_margin) / churn_rate
            
            return round(clv, 2)

            return 2400.0  # Default fallback value
