    async def generate_report(self, export_request: ExportRequest) -> Dict[str, Any]:
        """Generate comprehensive report in specified format"""
        try:
            # Store request
            self.export_requests[export_request.id] = export_request
            export_request.status = "processing"
            export_request.created_at = datetime.now()
            
            # Get session data
            session_data = await self._get_session_data(export_request.session_id)
            
            # Get branding configuration
            branding = self.branding_configs.get(export_request.user_id)
            
            # Generate report based on format
            if export_request.export_format == ExportFormat.PDF:
                file_path = await self.pdf_generator.generate_pdf_report(
                    session_data, export_request, branding
                )
            elif export_request.export_format == ExportFormat.POWERPOINT:
                file_path = await self.presentation_generator.generate_powerpoint(
                    session_data, export_request, branding
                )
            elif export_request.export_format == ExportFormat.EXCEL:
                file_path = await self.data_exporter.generate_excel_report(
                    session_data, export_request
                )
            elif export_request.export_format == ExportFormat.VIDEO:
                file_path = await self.video_exporter.generate_video_package(
                    session_data, export_request
                )
            else:
                raise ValueError(f"Unsupported export format: {export_request.export_format}")
            
            # Update request status
            export_request.status = "completed"
            export_request.completed_at = datetime.now()
            export_request.file_path = file_path
            export_request.file_size = os.path.getsize(file_path) if file_path else 0
            
            return {
                'success': True,
                'export_id': export_request.id,
                'file_path': file_path,
                'file_size': export_request.file_size
            }
            
        except Exception as e:
            export_request.status = "failed"
            self.logger.error(f"Error generating report: {e}")
            return {'success': False, 'error': str(e)}

class PDFReportGenerator:
    """PDF report generation with professional layouts"""
    
    def __init__(self):
        self.styles = getSampleStyleSheet()
        self._setup_custom_styles()
        
    def _setup_custom_styles(self):
        """Setup custom paragraph styles"""
        self.styles.add(ParagraphStyle(
            name='CustomTitle',
            parent=self.styles['Heading1'],
            fontSize=24,
            spaceAfter=30,
            textColor=colors.HexColor('#1976d2')
        ))
        
        self.styles.add(ParagraphStyle(
            name='CustomHeading',
            parent=self.styles['Heading2'],
            fontSize=16,
            spaceAfter=12,
            textColor=colors.HexColor('#424242')
        ))
    
    async def generate_pdf_report(self, session_data: Dict, export_request: ExportRequest, 
                                branding: Optional[BrandingConfig]) -> str:
        """Generate comprehensive PDF report"""
        try:
            # Create temporary file
            output_path = f"/tmp/report_{export_request.id}.pdf"
            
            # Create PDF document
            doc = SimpleDocTemplate(output_path, pagesize=A4)
            story = []
            
            # Title page
            await self._add_title_page(story, session_data, branding)
            
            # Executive summary
            await self._add_executive_summary(story, session_data)
            
            # Tactical analysis section
            if export_request.report_type == ReportType.TACTICAL_ANALYSIS:
                await self._add_tactical_analysis_section(story, session_data)
            
            # Performance metrics section
            await self._add_performance_metrics_section(story, session_data)
            
            # Visual analysis section
            await self._add_visual_analysis_section(story, session_data)
            
            # Recommendations section
            await self._add_recommendations_section(story, session_data)
            
            # Build PDF
            doc.build(story)
            
            return output_path
            
        except Exception as e:
            logging.error(f"Error generating PDF report: {e}")
            raise
    
    async def _add_title_page(self, story: List, session_data: Dict, 
                            branding: Optional[BrandingConfig]):
        """Add title page to PDF report"""
        try:
            # Add logo if available
            if branding and branding.logo_path:
                logo = Image(branding.logo_path, width=2*inch, height=1*inch)
                story.append(logo)
                story.append(Spacer(1, 20))
            
            # Add title
            title = session_data.get('match_info', {}).get('title', 'Match Analysis Report')
            story.append(Paragraph(title, self.styles['CustomTitle']))
            story.append(Spacer(1, 20))
            
            # Add match details
            match_info = session_data.get('match_info', {})
            details_data = [
                ['Date:', match_info.get('date', 'N/A')],
                ['Teams:', f"{match_info.get('home_team', 'Team A')} vs {match_info.get('away_team', 'Team B')}"],
                ['Venue:', match_info.get('venue', 'N/A')],
                ['Duration:', f"{match_info.get('duration', 90)} minutes"]
            ]
            
            details_table = Table(details_data, colWidths=[1.5*inch, 4*inch])
            details_table.setStyle(TableStyle([
                ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
                ('FONTNAME', (0, 0), (0, -1), 'Helvetica-Bold'),
                ('FONTSIZE', (0, 0), (-1, -1), 12),
                ('BOTTOMPADDING', (0, 0), (-1, -1), 8),
            ]))
            
            story.append(details_table)
            story.append(Spacer(1, 40))
            
        except Exception as e:
            logging.error(f"Error adding title page: {e}")
    
    async def _add_tactical_analysis_section(self, story: List, session_data: Dict):
        """Add tactical analysis section to PDF"""
        try:
            story.append(Paragraph("Tactical Analysis", self.styles['CustomHeading']))
            story.append(Spacer(1, 12))
            
            # Formation analysis
            formations = session_data.get('formations', [])
            if formations:
                dominant_formation = max(formations, key=lambda f: f.get('confidence', 0))
                
                formation_text = f"""
                <b>Dominant Formation:</b> {dominant_formation.get('formation_type', 'N/A')}<br/>
                <b>Confidence:</b> {dominant_formation.get('confidence', 0) * 100:.1f}%<br/>
                <b>Compactness:</b> {dominant_formation.get('compactness', 0):.2f}<br/>
                <b>Width:</b> {dominant_formation.get('width', 0):.2f}m<br/>
                <b>Depth:</b> {dominant_formation.get('depth', 0):.2f}m
                """
                
                story.append(Paragraph(formation_text, self.styles['Normal']))
                story.append(Spacer(1, 20))
            
            # Add tactical insights
            insights = session_data.get('tactical_insights', [])
            if insights:
                story.append(Paragraph("Key Tactical Insights:", self.styles['Heading3']))
                for insight in insights[:5]:  # Top 5 insights
                    story.append(Paragraph(f"• {insight}", self.styles['Normal']))
                story.append(Spacer(1, 20))
                
        except Exception as e:
            logging.error(f"Error adding tactical analysis section: {e}")

    def _flatten_dict(self, d: Dict, parent_key: str = '', sep: str = '_') -> Dict:
        """Flatten nested dictionary"""
        items = []
        for k, v in d.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, dict):
                items.extend(self._flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
        return dict(items)

                'active': True,

                'last_triggered': None,
                'success_count': 0,
                'failure_count': 0
            }

            return {'success': True, 'webhook_id': webhook_id}
            
        except Exception as e:

            return {'success': False, 'error': str(e)}
    
    async def trigger_webhook(self, event_type: str, payload: Dict) -> Dict[str, Any]:
        """Trigger webhooks for specific event"""
        try:
            triggered_count = 0
            
            for webhook_id, webhook in self.webhooks.items():
                if (webhook['active'] and event_type in webhook['events']):
                    success = await self._send_webhook(webhook, event_type, payload)
                    
                    if success:
                        webhook['success_count'] += 1
                        triggered_count += 1
                    else:
                        webhook['failure_count'] += 1
                    
                    webhook['last_triggered'] = datetime.now().isoformat()
            
            return {'success': True, 'triggered_count': triggered_count}
            
        except Exception as e:
            logging.error(f"Error triggering webhooks: {e}")
            return {'success': False, 'error': str(e)}
    
    async def _send_webhook(self, webhook: Dict, event_type: str, payload: Dict) -> bool:
        """Send webhook HTTP request"""
        try:
            import aiohttp
            
            webhook_payload = {
                'event': event_type,
                'timestamp': datetime.now().isoformat(),
                'data': payload
            }
            
            headers = {'Content-Type': 'application/json'}
            
            # Add signature if secret is provided
            if webhook.get('secret'):
                import hmac
                import hashlib
                
                signature = hmac.new(
                    webhook['secret'].encode(),
                    json.dumps(webhook_payload).encode(),
                    hashlib.sha256
                ).hexdigest()
                
                headers['X-Webhook-Signature'] = f"sha256={signature}"
            
            async with aiohttp.ClientSession() as session:
                async with session.post(
                    webhook['url'],
                    json=webhook_payload,
                    headers=headers,
                    timeout=aiohttp.ClientTimeout(total=30)
                ) as response:
                    return 200 <= response.status < 300
                    
        except Exception as e:
            logging.error(f"Error sending webhook: {e}")
            return False

            body = email_template['body']

            # Add attachment
            with open(report_path, "rb") as attachment:
                part = MIMEBase('application', 'octet-stream')
                part.set_payload(attachment.read())
                encoders.encode_base64(part)
                
                filename = os.path.basename(report_path)
                part.add_header(
                    'Content-Disposition',
                    f'attachment; filename= {filename}'
                )
                
                msg.attach(part)
            
            # Send email
            server = smtplib.SMTP(self.smtp_config['host'], self.smtp_config['port'])
            if self.smtp_config.get('use_tls'):
                server.starttls()
            
            server.login(self.smtp_config['username'], self.smtp_config['password'])
            server.send_message(msg)
            server.quit()
            
            return {'success': True, 'recipients_count': len(recipients)}
            
        except Exception as e:
            logging.error(f"Error sending email report: {e}")

    async def _calculate_next_send_time(self, frequency: str, time: str) -> str:
        """Calculate next send time based on frequency"""
        try:
            from datetime import datetime, timedelta
            import pytz
            
            now = datetime.now(pytz.UTC)
            
            if frequency == 'daily':
                next_send = now + timedelta(days=1)
            elif frequency == 'weekly':
                days_ahead = 7 - now.weekday()
                next_send = now + timedelta(days=days_ahead)
            elif frequency == 'monthly':
                # Next month, same day
                if now.month == 12:
                    next_send = now.replace(year=now.year + 1, month=1)
                else:
                    next_send = now.replace(month=now.month + 1)
            else:
                next_send = now + timedelta(days=1)  # Default to daily
            
            # Set specific time
            hour, minute = map(int, time.split(':'))
            next_send = next_send.replace(hour=hour, minute=minute, second=0, microsecond=0)
            
            return next_send.isoformat()
            
        except Exception as e:
            logging.error(f"Error calculating next send time: {e}")
            return (datetime.now() + timedelta(days=1)).isoformat()

                        )}
                      </Box>
                    }
                  />
                  <ListItemSecondaryAction>
                    <Box display="flex" alignItems="center" gap={1}>
                      <FormControlLabel
                        control={
                          <Switch
                            checked={schedule.active}
                            size="small"
                            onChange={(e) => {
                              // Update schedule active status
                            }}
                          />
                        }
                        label=""
                      />
                      <IconButton size="small">
                        <EditIcon />
                      </IconButton>
                      <IconButton size="small" color="error">
                        <DeleteIcon />
                      </IconButton>
                    </Box>
                  </ListItemSecondaryAction>
                </ListItem>
              ))}
            </List>
          </CardContent>
        </Card>
      </Grid>
    </Grid>
  );

  const renderIntegrationsTab = () => (
    <Grid container spacing={3}>
      <Grid item xs={12}>
        <Typography variant="h5" gutterBottom>Third-Party Integrations</Typography>
      </Grid>

      <Grid item xs={12} md={6}>
        <Card>
          <CardContent>
            <Typography variant="h6" gutterBottom>Available Integrations</Typography>
            <List>
              {integrations.map((integration) => (
                <ListItem key={integration.id}>
                  <ListItemText
                    primary={integration.name}
                    secondary={integration.provider}
                  />
                  <ListItemSecondaryAction>
                    <Chip
                      label={integration.connected ? 'Connected' : 'Not Connected'}
                      color={integration.connected ? 'success' : 'default'}
                      size="small"
                    />
                    <Button
                      size="small"
                      variant={integration.connected ? 'outlined' : 'contained'}
                      sx={{ ml: 1 }}
                    >
                      {integration.connected ? 'Disconnect' : 'Connect'}
                    </Button>
                  </ListItemSecondaryAction>
                </ListItem>
              ))}
            </List>
          </CardContent>
        </Card>
      </Grid>

      <Grid item xs={12} md={6}>
        <Card>
          <CardContent>
            <Typography variant="h6" gutterBottom>API Access</Typography>
            
            <Alert severity="info" sx={{ mb: 2 }}>
              Use our REST API to integrate Trackball data with your existing tools
            </Alert>

            <Box mb={2}>
              <Typography variant="subtitle2" gutterBottom>API Endpoint</Typography>
              <TextField
                fullWidth
                value="https://api.trackball.com/v1"
                variant="outlined"
                size="small"
                InputProps={{ readOnly: true }}
              />
            </Box>

            <Box mb={2}>
              <Typography variant="subtitle2" gutterBottom>Your API Key</Typography>
              <TextField
                fullWidth
                value="tk_1234567890abcdef"
                variant="outlined"
                size="small"
                type="password"
                InputProps={{ readOnly: true }}
              />
            </Box>

            <Button variant="outlined" startIcon={<APIIcon />}>
              View API Documentation
            </Button>
          </CardContent>
        </Card>
      </Grid>
    </Grid>
  );

  return (
    <Box sx={{ p: 3 }}>
      <Box display="flex" justifyContent="between" alignItems="center" mb={3}>
        <Typography variant="h4" component="h1">
          Export & Integration
        </Typography>
        <IntegrationIcon fontSize="large" color="primary" />
      </Box>

      <Tabs value={activeTab} onChange={(e, newValue) => setActiveTab(newValue)} sx={{ mb: 3 }}>
        <Tab label="Exports" icon={<ExportIcon />} />
        <Tab label="Scheduling" icon={<ScheduleIcon />} />
        <Tab label="Integrations" icon={<IntegrationIcon />} />
        <Tab label="Templates" icon={<PreviewIcon />} />
      </Tabs>

      {activeTab === 0 && renderExportTab()}
      {activeTab === 1 && renderSchedulingTab()}
      {activeTab === 2 && renderIntegrationsTab()}

      {/* Export Dialog */}
      <Dialog open={exportDialogOpen} onClose={() => setExportDialogOpen(false)} maxWidth="md" fullWidth>
        <DialogTitle>Create New Export</DialogTitle>
        <DialogContent>
          <Grid container spacing={2}>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>Session</InputLabel>
                <Select
                  value={newExport.sessionId}
                  onChange={(e) => setNewExport(prev => ({ ...prev, sessionId: e.target.value }))}
                >
                  <MenuItem value="session1">Match vs Team A</MenuItem>
                  <MenuItem value="session2">Training Session #45</MenuItem>
                  <MenuItem value="session3">Match vs Team B</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>Export Format</InputLabel>
                <Select
                  value={newExport.exportFormat}
                  onChange={(e) => setNewExport(prev => ({ ...prev, exportFormat: e.target.value }))}
                >
                  <MenuItem value="pdf">PDF Report</MenuItem>
                  <MenuItem value="powerpoint">PowerPoint Presentation</MenuItem>
                  <MenuItem value="excel">Excel Spreadsheet</MenuItem>
                  <MenuItem value="video">Video Package</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12}>
              <FormControl fullWidth>
                <InputLabel>Report Type</InputLabel>
                <Select
                  value={newExport.reportType}
                  onChange={(e) => setNewExport(prev => ({ ...prev, reportType: e.target.value }))}
                >
                  <MenuItem value="tactical_analysis">Tactical Analysis</MenuItem>
                  <MenuItem value="performance_summary">Performance Summary</MenuItem>
                  <MenuItem value="match_report">Match Report</MenuItem>
                  <MenuItem value="player_profile">Player Profile</MenuItem>
                </Select>
              </FormControl>
            </Grid>
          </Grid>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setExportDialogOpen(false)}>Cancel</Button>
          <Button 
            onClick={handleExport} 
            variant="contained"
            disabled={!newExport.sessionId || !newExport.exportFormat || !newExport.reportType}
          >
            Create Export
          </Button>
        </DialogActions>
      </Dialog>

      {/* Schedule Dialog */}
      <Dialog open={scheduleDialogOpen} onClose={() => setScheduleDialogOpen(false)} maxWidth="md" fullWidth>
        <DialogTitle>Create Delivery Schedule</DialogTitle>
        <DialogContent>
          <Grid container spacing={2}>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="Schedule Name"
                value={newSchedule.name}
                onChange={(e) => setNewSchedule(prev => ({ ...prev, name: e.target.value }))}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>Report Type</InputLabel>
                <Select
                  value={newSchedule.reportType}
                  onChange={(e) => setNewSchedule(prev => ({ ...prev, reportType: e.target.value }))}
                >
                  <MenuItem value="weekly_summary">Weekly Summary</MenuItem>
                  <MenuItem value="match_reports">Match Reports</MenuItem>
                  <MenuItem value="performance_trends">Performance Trends</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>Frequency</InputLabel>
                <Select
                  value={newSchedule.frequency}
                  onChange={(e) => setNewSchedule(prev => ({ ...prev, frequency: e.target.value }))}
                >
                  <MenuItem value="daily">Daily</MenuItem>
                  <MenuItem value="weekly">Weekly</MenuItem>
                  <MenuItem value="monthly">Monthly</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="Recipients (comma-separated emails)"
                value={newSchedule.recipients}
                onChange={(e) => setNewSchedule(prev => ({ ...prev, recipients: e.target.value }))}
                placeholder="coach@team.com, analyst@team.com"
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Delivery Time"
                type="time"
                value={newSchedule.time}
                onChange={(e) => setNewSchedule(prev => ({ ...prev, time: e.target.value }))}
                InputLabelProps={{ shrink: true }}
              />
            </Grid>
          </Grid>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setScheduleDialogOpen(false)}>Cancel</Button>
          <Button 
            onClick={handleCreateSchedule} 
            variant="contained"
            disabled={!newSchedule.name || !newSchedule.reportType || !newSchedule.recipients}
          >
            Create Schedule
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
};
